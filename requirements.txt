# VisionMate — Python packages for the Raspberry Pi 5 (Raspberry Pi OS 64-bit, Python 3.11)
#
# Install the system packages first:
#   sudo apt-get install mpg123 tesseract-ocr cmake build-essential libopenblas-dev liblapack-dev
# Then, inside a virtual environment:
#   pip install -r requirements.txt
#
# face_recognition needs dlib, which compiles from source (see Instructions.txt if pip fails).
# requirements-dev-freeze.txt is the full pip freeze of the original development machine,
# kept for reference only.

numpy<2                 # mediapipe needs NumPy 1.x
opencv-python
tensorflow>=2.15        # older Keras model files are rebuilt in code, so Keras 3 works
h5py
ultralytics             # YOLOv8n (object detection)
pytesseract             # OCR (needs the tesseract-ocr system package)
face_recognition        # dlib face encodings
cvzone                  # hand tracking wrapper
mediapipe               # hand landmarks
scikit-learn==1.5.2      # random forests; the shipped sign models were saved with 1.5 (any version after retraining)
joblib
pandas
gTTS                    # speech (online service; needs an internet connection)
matplotlib              # evaluation figures
